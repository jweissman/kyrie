it('K2 asleep', async () => {
  process.chdir(R);
  const s = await PlaySession.start('test/fixtures/probes/sleep-hours/probe.campaign.json', 'q', NodeBundleSource.shared);
  const { ScreenModel } = await import(`${R}/src/ui/screen-model`); const { ChoiceList } = await import(`${R}/src/ui/choice-list`);
  const leaves = (l: any[], t = ''): any[] => l.flatMap(c => c.children ? leaves(c.children, `${t}${c.label} › `) : [{ label: t + c.label, command: c.command }]);
  const c = leaves(ChoiceList.all(ChoiceList.explore(ScreenModel.build(s.campaign)))).find(x => /^Wait › A while/i.test(x.label));
  await s.perform(c.command);
  const r = s.campaign.readout, body = r.creatures.find((x: any) => r.partyIds.includes(x.id));
  console.log('ASLEEP:', body.state.conditions.some((x: any) => x.id === 'asleep'));
}, 120000);
